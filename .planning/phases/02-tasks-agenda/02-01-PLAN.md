    ```ts
    export type Priority = "high" | "medium" | "low";

    export interface Task {
      id: number;
      title: string;
      description: string | null;
      priority: Priority;
      due_date: string | null;      // ISO 8601 from API
      reminder_at: string | null;   // ISO 8601
      recurrence_cron: string | null;
      completed: boolean;
      created_at: string;
      updated_at: string;
    }

    export interface TaskCreate {
      title: string;
      description?: string | null;
      priority?: Priority;
      due_date?: string | null;
      reminder_at?: string | null;
      recurrence_cron?: string | null;
    }

    // Agenda merge model (Today view)
    export interface AgendaItem {
      id: string;
      title: string;
      time: string | null;   // "09:00" 24h, or null for all-day
      priority: Priority | null;
      isEvent: boolean;       // true = placeholder calendar event
    }
    ```
