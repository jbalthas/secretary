    ```python
    import httpx
    from app.config import settings

    PUSHOVER_URL = "https://api.pushover.net/1/messages.json"


    class PushoverClient:
        def send(self, title: str, message: str, priority: int = 0) -> None:
            with httpx.Client(timeout=10) as client:
                r = client.post(PUSHOVER_URL, data={
                    "token": settings.pushover_api_token,
                    "user": settings.pushover_user_key,
                    "title": title,
                    "message": message or " ",
                    "priority": priority,
                })
                r.raise_for_status()
    ```

  <action>
    Create backend/app/scheduler.py implementing this exact contract (per D-04, D-05, D-06, D-07, D-08; sync URL per db.py precedent; misfire handling per Pitfall 5; sync job callable per Pitfall 1; JobLookupError swallow per Pitfall 3):

    ```python
    from apscheduler.schedulers.asyncio import AsyncIOScheduler
    from apscheduler.jobstores.sqlalchemy import SQLAlchemyJobStore
    from app.config import settings

    _sync_url = settings.database_url.replace("+aiosqlite", "")

    scheduler = AsyncIOScheduler(
        jobstores={"default": SQLAlchemyJobStore(url=_sync_url)},
        timezone="UTC",
    )


    def _pushover_priority(priority) -> int:
        return 1 if priority is not None and priority.value == "high" else 0


    def _send_reminder(title: str, body: str, priority: int) -> None:
        from app.services.pushover import PushoverClient
        PushoverClient().send(title=title, message=body, priority=priority)


    def upsert_reminder(task) -> None:
        if task.reminder_at is None:
            remove_reminder(task.id)
            return
        scheduler.add_job(
            _send_reminder,
            "date",
            run_date=task.reminder_at,
            id=f"reminder_task_{task.id}",
            replace_existing=True,
            misfire_grace_time=3600,
            kwargs={
                "title": task.title,
                "body": task.description or "",
                "priority": _pushover_priority(task.priority),
            },
        )


    def remove_reminder(task_id: int) -> None:
        try:
            scheduler.remove_job(f"reminder_task_{task_id}")
        except Exception:
            pass
    ```

    Critical details:
    - `_send_reminder` is a plain `def` (NOT async) — APScheduler 3.x runs it in a thread pool and will not await a coroutine (Pitfall 1).
    - The PushoverClient import is deferred inside `_send_reminder` to avoid circular import at module load.
    - `misfire_grace_time=3600` lets a reminder fire up to 1h late after a Pi reboot (Pitfall 5 — required for the gate test).
    - Job id is `reminder_task_{task_id}` (D-07); `replace_existing=True` makes upsert idempotent (D-04).
    - `remove_reminder` swallows JobLookupError so removing an already-fired or never-scheduled job is safe (D-06, Pitfall 3).

    ```python
    from contextlib import asynccontextmanager
    from fastapi import FastAPI
    from app.config import settings
    from app.routers import tasks
    from app.scheduler import scheduler


    @asynccontextmanager
    async def lifespan(app: FastAPI):
        scheduler.start()
        yield
        scheduler.shutdown()


    app = FastAPI(title="My Secretary", lifespan=lifespan)

    app.include_router(tasks.router)


    @app.get(f"{settings.api_prefix}/health")
    async def health():
        return {"status": "ok"}
    ```
