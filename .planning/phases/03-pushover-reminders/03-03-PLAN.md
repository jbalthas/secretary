@router.post("/", response_model=TaskRead, status_code=201)
async def create_task(body, session):
    task = Task(**body.model_dump())
    session.add(task); await session.commit(); await session.refresh(task)
    return task

@router.patch("/{task_id}", response_model=TaskRead)
async def update_task(task_id, body, session):
    task = await session.get(Task, task_id)
    if not task: raise HTTPException(404)
    for k, v in body.model_dump(exclude_unset=True).items(): setattr(task, k, v)
    await session.commit(); await session.refresh(task)
    return task

@router.delete("/{task_id}", status_code=204)
async def delete_task(task_id, session):
    task = await session.get(Task, task_id)
    if not task: raise HTTPException(404)
    await session.delete(task); await session.commit()
