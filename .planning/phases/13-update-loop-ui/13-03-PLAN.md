    Create `frontend/src/hooks/useUpdate.ts` — a thin hook wrapping the resolve POST. It does NOT own UI state (Today owns the phase/candidate state); it just performs the network calls and returns typed results + an error flag:
    ```typescript
    export function useUpdate() {
      async function submit(text: string): Promise<UpdateResponse | null> {
        const res = await fetch("/api/v1/updates/resolve", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ text }),
        });
        return res.ok ? (await res.json()) as UpdateResponse : null;
      }
      async function confirm(text: string, c: UpdateCandidate, action: string): Promise<UpdateResponse | null> {
        const res = await fetch("/api/v1/updates/resolve", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ text, confirmed_id: c.entity_id, confirmed_type: c.entity_type, confirmed_action: action }),
        });
        return res.ok ? (await res.json()) as UpdateResponse : null;
      }
      return { submit, confirm };
    }
    ```
    (Default `action` to `"done"` at the call site — confirm path applies done unless the original parse said otherwise; v2.1 keeps it simple as "done".)
